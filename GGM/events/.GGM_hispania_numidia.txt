country_event = { #Become Numidia after losing Hispania
	id = 49958 
	title = "EVTNAME49958" 
	desc = "EVTDESC49958"
	#picture = "building_temple"

	trigger = {
		tag = HNU
		OR = {
			AND = {
				exists = HIS
				NOT = { is_our_vassal = HIS }
			}
			NOT = { HIS = { any_core = { owned_by = HNU } } }
		}
	}
	
	mean_time_to_happen = {
		months = 1
	}
	
	option = {
		name = "EVT49958OPTA"
		change_tag = NUM
		remove_accepted_culture = hispanic
        HIS = { all_core = { remove_core = NUM } }
	}
	
}
