# Internatinal sanctions
country_event = {
	id = 50057
	title = "EVTNAME50057"
	desc = "EVTDESC50057"
	
    trigger = {
        ai = no
        badboy = 1.3 #OVER 36.4 INFAMY
        is_disarmed = no
		NOT = { has_global_flag = sanctions_disabled }
        NOT = {
            has_country_modifier = international_sanctions
        }
    }

    mean_time_to_happen = {
        months = 1
    }
	
	option = {
		name = "EVT50057OPTA"
        prestige = -20
        add_country_modifier = {
            name = international_sanctions
            duration = 36525
        }
	}
}
