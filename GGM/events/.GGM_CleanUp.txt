country_event = {
	id = 49994 
	title = "EVTNAME49994" 
	desc = "EVTDESC49994"
	#picture = "independence_dec"

	is_triggered_only = yes
	
	option = {
		name = "Yes, we will take this state."
		
		random_country = {
			limit = {
				has_country_flag = theft_victim
			}
			random_state = {
				limit = {
					any_owned_province = {
						has_province_modifier = annexation_candidate
					}
				}
				
			}
			
			any_owned = {
				limit = {
					has_province_modifier = annexation_candidate
				}
				secede_province = THIS
			}
			
			any_owned = {
				limit = {
					has_province_modifier = not_interested
				}
				remove_province_modifier = not_interested
			}
			clr_country_flag = theft_victim
			clr_country_flag = theft_setup
		
		}
		
		any_owned = {
			limit = {
				has_province_modifier = annexation_candidate
			}
			remove_province_modifier = annexation_candidate
		}
	
	}
	
	option = {
		name = "No, this is not the state we want."
		
		random_country = {
			limit = {
				has_country_flag = theft_victim
			}
			any_owned = {
				limit = {
					has_province_modifier = annexation_candidate
				}
				remove_province_modifier = annexation_candidate
				add_province_modifier = {
					name = not_interested
					duration = 365
				}
			}
			
			random_state = {
				limit = {
					any_owned_province = {
						NOT = {
							has_province_modifier = not_interested
						}
						OR = {
							NOT = {
								port = no
							}
							any_neighbor_province = {
								owned_by = THIS
							}
						}
					}
					
				
				}
				
				any_owned = {
					add_province_modifier = {
						name = annexation_candidate
						duration = 365
					}
				}
				
			
			}
		
		}
		
		THIS = {
			country_event = {
				id = 49994
				days = 3
			
			}
		}
		
	}

}

country_event = {
	id = 49974 
	title = "EVTNAME49974" 
	desc = "EVTDESC49974"
	
	allow_multiple_instances = no

	trigger = {
		ai = yes
		OR = {
			any_owned_province = {
				OR = {
					is_core = CAP
					is_core = CLT
				}
				any_core = {
					NOT = {
						tag = CAP
						tag = CLT
					}
				}
			}
			any_neighbor_country = {
				any_owned_province = {
					OR = {
						is_core = CAP
						is_core = CLT
					}
					any_core = {
						NOT = {
							tag = CAP
							tag = CLT
						}
					}
				}
			
			}
			any_greater_power = {
				any_owned_province = {
					OR = {
						is_core = CAP
						is_core = CLT
					}
					any_core = {
						NOT = {
							tag = CAP
							tag = CLT
						}
					}
				}
			
			}
		
		}
	}
	
	mean_time_to_happen = {
		months = 1
	
	}
	
	option = {
		name = "Colonial Cores Cleanup Fired"
	
		any_country = {
			limit = {
				any_owned_province = {
					OR = {
						is_core = CAP
						is_core = CLT
					}
					any_core = {
						NOT = {
							tag = CAP
							tag = CLT
						}
					}
				}
			}
			
			any_owned = {
				limit = {
					AND = {
						OR = {
							is_core = CAP
							is_core = CLT
						}
						any_core = {
							NOT = {
								tag = CAP
								tag = CLT
							}
						}
					}
				
				}
				remove_core = CAP
				remove_core = CLT
			
			}
		
		}
	
	}
	
	
}
