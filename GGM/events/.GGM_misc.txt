# Gain Core event
province_event = {
	id = 49956
	title = "EVTNAME49956"
	desc = "EVTDESC49956"
	
	trigger = {
		NOT = { is_core = THIS }
		controlled_by = THIS
		is_colonial = no
		OR = {
			is_primary_culture = yes
			is_accepted_culture = yes
			any_neighbor_province = {
				AND = {
					is_core = THIS
					controlled_by = THIS
				}
			}
		}
		owner = {
			tag = THIS
			is_greater_power = yes
			nationalism_n_imperialism = 1
		}
	}
	
	mean_time_to_happen = {
		months = 360 #180
		modifier = {
			factor = 0.5
			is_primary_culture = yes
		}
		modifier = {
			factor = 0.8
			is_coastal = yes
		}
		modifier = {
			factor = 0.9
			NOT = {
				average_militancy = 1
			}
		}
		modifier = {
			factor = 0.9
			NOT = {
				average_militancy = 2
			}
		}
		modifier = {
			factor = 0.95
			NOT = {
				average_militancy = 3
			}
		}
		modifier = {
			factor = 0.9
			NOT = {
				average_consciousness = 4
			}
		}
	}
	
	option = {
		name = "EVT49956OPTA"
		add_core = THIS
	}
}

# AI_WAR_OF_OPPERTUNITY
country_event = {
	id = 123123
	title = "EVTNAME123123"
	desc = "EVTDESC123123"
	
	is_triggered_only = yes

   
	
	option = {
		name = "EVT123123OPTA"

         if = {
        limit = {
            FROM = {
                any_owned_province = {
                    is_core = THIS
                }
            }
        }

        FROM = {
			add_casus_belli = {
				target = THIS
				type = acquire_all_cores
				months = 12
			}
		}
		
		THIS = {
			war = {
				target = FROM
				attacker_goal = { casus_belli = acquire_all_cores }
				call_ally = yes
			}
            set_country_flag = cb_check_done
		}

    }

        if = {
            limit = {
                THIS = {
                    NOT = {
                        has_country_flag = cb_check_done
                    }
                }

            }
		    FROM = {
			    add_casus_belli = {
				    target = THIS
				    type = cut_down_to_size
				    months = 12
			    }
		    }
		
		    THIS = {
			    war = {
				    target = FROM
				    attacker_goal = { casus_belli = cut_down_to_size }
				    call_ally = yes
			    }
		    }
        }
        clr_country_flag = cb_check_done
	}
}
